datos = [5, 8, 2, 1]

# Método de la burbuja
for x in range(len(datos) - 1):
    for y in range(len(datos) - 1 - x):
        if datos[y] < datos[y + 1]:
            # Intercambio (swap)
            burbuja = datos[y]
            datos[y] = datos[y + 1]
            datos[y + 1] = burbuja
print("Datos ordenados:", datos)
